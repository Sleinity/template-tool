export type DeferredFieldCommitResult = { accepted: boolean; stale?: boolean; blockingErrors?: readonly { message: string }[] };

export type DeferredFieldDraft = {
  valid: boolean;
  message?: string;
  focus?: () => void;
  commit: () => DeferredFieldCommitResult | Promise<DeferredFieldCommitResult>;
};

export type DeferredFieldFlushResult = { accepted: boolean; invalidKeys: string[] };

type Entry = DeferredFieldDraft & { generation: number; timer?: ReturnType<typeof setTimeout>; running?: Promise<boolean>; runningGeneration?: number };

/** Host-owned typing coordinator. Canonical sessions remain immediate when commit is invoked. */
export class DeferredFieldDraftCoordinator {
  private entries = new Map<string, Entry>();
  private listeners = new Set<() => void>();
  private revision = 0;

  constructor(readonly delayMs: number) {}

  subscribe = (listener: () => void) => { this.listeners.add(listener); return () => this.listeners.delete(listener); };
  getSnapshot = () => this.revision;
  get pendingCount() { return this.entries.size; }
  get invalidCount() { return [...this.entries.values()].filter((entry) => !entry.valid).length; }
  get dirty() { return this.entries.size > 0; }

  private notify() { this.revision += 1; this.listeners.forEach((listener) => listener()); }

  schedule(key: string, draft: DeferredFieldDraft) {
    const previous = this.entries.get(key);
    if (previous?.timer) clearTimeout(previous.timer);
    const entry: Entry = { ...draft, generation: (previous?.generation ?? 0) + 1, running: previous?.running, runningGeneration: previous?.runningGeneration };
    this.entries.set(key, entry);
    if (entry.valid) entry.timer = setTimeout(() => { void this.flushKey(key); }, this.delayMs);
    this.notify();
  }

  cancel(key: string) {
    const entry = this.entries.get(key);
    if (!entry) return;
    if (entry.timer) clearTimeout(entry.timer);
    this.entries.delete(key);
    this.notify();
  }

  async flushKey(key: string): Promise<boolean> {
    const entry = this.entries.get(key);
    if (!entry) return true;
    if (!entry.valid) { entry.focus?.(); return false; }
    if (entry.timer) { clearTimeout(entry.timer); entry.timer = undefined; }
    if (entry.running && entry.runningGeneration === entry.generation) return entry.running;
    if (entry.running) { await entry.running; return this.flushKey(key); }
    const generation = entry.generation;
    const running = Promise.resolve().then(entry.commit).then((result) => {
      const accepted = result.accepted && !result.stale;
      const current = this.entries.get(key);
      if (current?.runningGeneration === generation) { current.running = undefined; current.runningGeneration = undefined; }
      if (current?.generation === generation) {
        if (accepted) this.entries.delete(key);
        else { current.valid = false; current.message = result.blockingErrors?.map((issue) => issue.message).join("\n") || "This value could not be applied."; current.focus?.(); }
      } else if (current?.valid && !current.timer) {
        current.timer = setTimeout(() => { void this.flushKey(key); }, this.delayMs);
      }
      this.notify();
      return accepted;
    }).catch(() => { const current = this.entries.get(key); if (current?.runningGeneration === generation) { current.running = undefined; current.runningGeneration = undefined; } if (current?.generation === generation) current.valid = false; this.notify(); return false; });
    entry.running = running;
    entry.runningGeneration = generation;
    return running;
  }

  async flush(): Promise<DeferredFieldFlushResult> {
    const invalidKeys: string[] = [];
    for (const key of [...this.entries.keys()]) if (!await this.flushKey(key)) invalidKeys.push(key);
    return { accepted: invalidKeys.length === 0, invalidKeys };
  }

  dispose() {
    for (const entry of this.entries.values()) if (entry.timer) clearTimeout(entry.timer);
    this.entries.clear();
    this.notify();
    this.listeners.clear();
  }
}

export const STUDIO_FIELD_DRAFT_DELAY_MS = 325;

export async function waitForCanonicalExportReadiness(session: {
  getSnapshot(): { readiness: { exportReady: boolean; state: string; reasons: readonly string[] } };
  subscribe(listener: () => void): () => void;
}, timeoutMs = 10000): Promise<void> {
  if (session.getSnapshot().readiness.exportReady) return;
  await new Promise<void>((resolve, reject) => {
    const timer = setTimeout(() => { stop(); reject(new Error("The preview did not become ready for export in time.")); }, timeoutMs);
    const inspect = () => {
      const readiness = session.getSnapshot().readiness;
      if (readiness.exportReady) { clearTimeout(timer); stop(); resolve(); }
      else if (readiness.state === "blocked") { clearTimeout(timer); stop(); reject(new Error(readiness.reasons[0] ?? "The current content is not ready to export.")); }
    };
    const stop = session.subscribe(inspect);
    inspect();
  });
}
